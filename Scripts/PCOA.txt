# ============================================================
# PRINCIPAL COORDINATE ANALYSIS (PCoA)
# Bray-Curtis distance
# ============================================================

# ============================================================
# DATA STRUCTURE
# ============================================================
#
# The input abundance table should have:
#
#   - Rows = samples/observations
#   - Columns = features/variables
#   - Row names = unique sample/observation identifiers
#   - Values = non-negative abundance measurements
#
# A separate grouping vector should contain one group label
# for each sample, in the same order as the abundance table.
#
# Example:
#
#              Feature_1  Feature_2  Feature_3
# Sample_01        10          5          2
# Sample_02         8          3          7
# Sample_03        15          9          1
#
# Group:
# Sample_01      Group_A
# Sample_02      Group_B
# Sample_03      Group_A
#
# ============================================================

library(vegan)
library(ggplot2)
library(dplyr)

# ============================================================
# USER INPUTS
# ============================================================

# Replace with the path to the abundance data.
abundance_data <- read.csv(
  "abundance_data.csv",
  header = TRUE,
  row.names = 1,
  check.names = FALSE
)

# Replace with the path to the metadata/group information.
metadata <- read.csv(
  "metadata.csv",
  header = TRUE,
  check.names = FALSE
)

# Name of the column containing sample/observation identifiers.
sample_id_column <- "Sample"

# Name of the column containing group labels.
group_column <- "Group"

# ============================================================
# PREPARE DATA
# ============================================================

# Extract sample identifiers and group information.
sample_ids <- metadata[[sample_id_column]]
group <- metadata[[group_column]]

# Ensure metadata and abundance data contain the same samples.
common_samples <- intersect(
  rownames(abundance_data),
  sample_ids
)

abundance_data <- abundance_data[common_samples, , drop = FALSE]

metadata <- metadata[
  match(common_samples, metadata[[sample_id_column]]),
  ,
  drop = FALSE
]

group <- metadata[[group_column]]

# Check that sample order is correctly matched.
stopifnot(
  identical(rownames(abundance_data), metadata[[sample_id_column]])
)

# ============================================================
# BRAY-CURTIS DISTANCE
# ============================================================

bray_dist <- vegdist(
  abundance_data,
  method = "bray"
)

# ============================================================
# PRINCIPAL COORDINATE ANALYSIS
# ============================================================

pcoa <- cmdscale(
  bray_dist,
  eig = TRUE,
  k = 2
)

# ============================================================
# VARIANCE EXPLAINED
# ============================================================

# Calculate the percentage of variation explained by each
# PCoA axis using the eigenvalues.
positive_eigenvalues <- pcoa$eig[pcoa$eig > 0]

var_exp <- 100 * positive_eigenvalues /
  sum(positive_eigenvalues)

pc1_percent <- round(var_exp[1], 2)
pc2_percent <- round(var_exp[2], 2)

# ============================================================
# PCoA COORDINATES
# ============================================================

pcoa_df <- data.frame(
  Sample = rownames(abundance_data),
  PC1 = pcoa$points[, 1],
  PC2 = pcoa$points[, 2],
  Group = group
)

# ============================================================
# SAVE PCoA COORDINATES
# ============================================================

write.csv(
  pcoa_df,
  "pcoa_coordinates.csv",
  row.names = FALSE
)

# ============================================================
# PLOT
# ============================================================

ggplot(
  pcoa_df,
  aes(
    x = PC1,
    y = PC2,
    color = Group,
    shape = Group
  )
) +
  geom_point(
    size = 4,
    alpha = 0.6
  ) +
  stat_ellipse(
    aes(fill = Group),
    geom = "polygon",
    alpha = 0.15,
    linetype = "dashed",
    level = 0.95
  ) +
  labs(
    title = "PCoA Based on Bray-Curtis Distance",
    x = paste0("PC1 (", pc1_percent, "%)"),
    y = paste0("PC2 (", pc2_percent, "%)")
  ) +
  theme_minimal() +
  theme(
    legend.title = element_blank()
  )
