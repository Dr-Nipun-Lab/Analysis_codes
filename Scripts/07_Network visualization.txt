# ============================================================
# Network visualization of significant cross-dataset
# correlations
#
# Input:
# - Correlation results generated from two datasets
# - Dataset 1 features are represented as one node type
# - Dataset 2 features are represented as another node type
#
# Edge:
# - Positive correlation = red
# - Negative correlation = blue
# - Edge width = absolute correlation strength
# ============================================================

library(igraph)

# ------------------------------------------------------------
# Input
# ------------------------------------------------------------

# Use the long-format correlation results generated by the
# correlation analysis script.

correlation_results <- read.csv(
  "correlation_results_long.csv",
  check.names = FALSE
)

# Expected columns:
# Feature_1
# Feature_2
# correlation
# p_value

# Significance threshold for network construction
p_threshold <- 0.05

# ------------------------------------------------------------
# Prepare significant correlations
# ------------------------------------------------------------

correlation_edges <- correlation_results[
  !is.na(correlation_results$p_value) &
    correlation_results$p_value < p_threshold &
    correlation_results$correlation != 0,
]

if (nrow(correlation_edges) == 0) {
  stop("No significant correlations were found.")
}

# ------------------------------------------------------------
# Edge properties
# ------------------------------------------------------------

correlation_edges$edge_color <- ifelse(
  correlation_edges$correlation > 0,
  adjustcolor("red", alpha.f = 0.45),
  adjustcolor("blue", alpha.f = 0.45)
)

correlation_edges$edge_width <-
  abs(correlation_edges$correlation) * 20

# ============================================================
# 1. NETWORK FOR A SINGLE FEATURE
# ============================================================

# Select the feature to visualize.
#
# Replace this with the desired feature name, or select
# interactively from the available features.

available_features <- unique(
  correlation_edges$Feature_1
)

feature_of_interest <- available_features[1]

cat(
  "Plotting network for feature:",
  feature_of_interest,
  "\n"
)

# ------------------------------------------------------------
# Select correlations involving the feature
# ------------------------------------------------------------

feature_edges <- correlation_edges[
  correlation_edges$Feature_1 == feature_of_interest,
]

if (nrow(feature_edges) == 0) {
  stop(
    "No significant correlations found for the selected feature."
  )
}

# ------------------------------------------------------------
# Create graph
# ------------------------------------------------------------

feature_graph <- graph_from_data_frame(
  feature_edges[, c(
    "Feature_1",
    "Feature_2",
    "correlation",
    "edge_color",
    "edge_width"
  )],
  directed = FALSE
)

# ------------------------------------------------------------
# Define node types
# ------------------------------------------------------------

V(feature_graph)$type <- ifelse(
  V(feature_graph)$name == feature_of_interest,
  "Feature_1",
  "Feature_2"
)

# ------------------------------------------------------------
# Node colours
# ------------------------------------------------------------

feature1_color <- adjustcolor(
  "#7EA8F8",
  alpha.f = 0.9
)

feature2_color <- adjustcolor(
  "#FFD97D",
  alpha.f = 0.9
)

V(feature_graph)$color <- ifelse(
  V(feature_graph)$type == "Feature_1",
  feature1_color,
  feature2_color
)

# ------------------------------------------------------------
# Node properties
# ------------------------------------------------------------

V(feature_graph)$size <- ifelse(
  V(feature_graph)$type == "Feature_1",
  25,
  12
)

V(feature_graph)$label.color <- "black"
V(feature_graph)$label.cex <- 0.85
V(feature_graph)$label.family <- "sans"
V(feature_graph)$frame.color <- NA

# ------------------------------------------------------------
# Plot
# ------------------------------------------------------------

set.seed(123)

par(
  bg = "#f7f7fb",
  family = "sans",
  mar = c(1, 1, 1, 1)
)

plot(
  feature_graph,
  layout = layout_with_fr(feature_graph),
  edge.color = E(feature_graph)$edge_color,
  edge.width = E(feature_graph)$edge_width,
  vertex.color = V(feature_graph)$color,
  vertex.size = V(feature_graph)$size,
  vertex.label = V(feature_graph)$name,
  vertex.label.color = V(feature_graph)$label.color,
  vertex.label.cex = V(feature_graph)$label.cex,
  vertex.label.family = "sans",
  vertex.frame.color = NA,
  main = paste(
    "Correlation Network for",
    feature_of_interest
  )
)

# ------------------------------------------------------------
# Legends
# ------------------------------------------------------------

usr <- par("usr")

x_leg <- usr[1] - 0.02 * diff(usr[1:2])

legend(
  x = x_leg,
  y = usr[3] + 0.65 * diff(usr[3:4]),
  legend = c("Dataset 1 feature", "Dataset 2 feature"),
  col = c(feature1_color, feature2_color),
  pch = 19,
  pt.cex = c(1.8, 1.3),
  bty = "n",
  title = "Node types",
  title.font = 2,
  xpd = NA
)

legend(
  x = x_leg,
  y = usr[3] + 0.35 * diff(usr[3:4]),
  legend = c(
    "Positive correlation",
    "Negative correlation"
  ),
  col = c(
    adjustcolor("red", alpha.f = 0.6),
    adjustcolor("blue", alpha.f = 0.6)
  ),
  lwd = 3,
  seg.len = 0.7,
  bty = "n",
  title = "Edge types",
  title.font = 2,
  xpd = NA
)

# ============================================================
# 2. UNIFIED NETWORK
# ============================================================

# ------------------------------------------------------------
# Create graph containing all significant correlations
# ------------------------------------------------------------

unified_graph <- graph_from_data_frame(
  correlation_edges[, c(
    "Feature_1",
    "Feature_2",
    "correlation",
    "edge_color",
    "edge_width"
  )],
  directed = FALSE
)

# ------------------------------------------------------------
# Define node types
# ------------------------------------------------------------

V(unified_graph)$type <- ifelse(
  V(unified_graph)$name %in%
    correlation_edges$Feature_1,
  "Feature_1",
  "Feature_2"
)

# ------------------------------------------------------------
# Node properties
# ------------------------------------------------------------

V(unified_graph)$color <- ifelse(
  V(unified_graph)$type == "Feature_1",
  feature1_color,
  feature2_color
)

V(unified_graph)$size <- ifelse(
  V(unified_graph)$type == "Feature_1",
  20,
  10
)

V(unified_graph)$label.color <- "black"

V(unified_graph)$label.cex <- ifelse(
  V(unified_graph)$type == "Feature_1",
  0.9,
  0.7
)

V(unified_graph)$label.family <- "sans"
V(unified_graph)$frame.color <- NA

# ------------------------------------------------------------
# Plot unified network
# ------------------------------------------------------------

set.seed(123)

par(
  bg = "#f7f7fb",
  family = "sans",
  mar = c(1, 1, 1, 1)
)

plot(
  unified_graph,
  layout = layout_with_fr(unified_graph),
  edge.color = E(unified_graph)$edge_color,
  edge.width = E(unified_graph)$edge_width,
  vertex.color = V(unified_graph)$color,
  vertex.size = V(unified_graph)$size,
  vertex.label = V(unified_graph)$name,
  vertex.label.color = V(unified_graph)$label.color,
  vertex.label.cex = V(unified_graph)$label.cex,
  vertex.label.family = "sans",
  vertex.frame.color = NA,
  main = "Network of Significant Correlations"
)

# ------------------------------------------------------------
# Legends
# ------------------------------------------------------------

usr <- par("usr")

x_leg <- usr[1] - 0.02 * diff(usr[1:2])

legend(
  x = x_leg,
  y = usr[3] + 0.65 * diff(usr[3:4]),
  legend = c(
    "Dataset 1 feature",
    "Dataset 2 feature"
  ),
  col = c(
    feature1_color,
    feature2_color
  ),
  pch = 19,
  pt.cex = c(1.6, 1.2),
  bty = "n",
  title = "Node types",
  title.font = 2,
  xpd = NA
)

legend(
  x = x_leg,
  y = usr[3] + 0.35 * diff(usr[3:4]),
  legend = c(
    "Positive correlation",
    "Negative correlation"
  ),
  col = c(
    adjustcolor("red", alpha.f = 0.6),
    adjustcolor("blue", alpha.f = 0.6)
  ),
  lwd = 3,
  seg.len = 0.7,
  bty = "n",
  title = "Edge types",
  title.font = 2,
  xpd = NA
)

# ------------------------------------------------------------
# Save network edge list
# ------------------------------------------------------------

write.csv(
  correlation_edges,
  "significant_correlation_edges.csv",
  row.names = FALSE
)
