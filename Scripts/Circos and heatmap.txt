# ============================================================
# Cross-dataset correlation analysis with heatmap and Circos
#
# Input structure:
# - Two CSV files
# - Rows = samples
# - Columns = numerical features
# - The same sample order must be present in both datasets
# ============================================================

library(ggplot2)
library(reshape2)
library(circlize)

# ------------------------------------------------------------
# Input data
# ------------------------------------------------------------

file1 <- file.choose()
file2 <- file.choose()

data1 <- read.csv(file1, row.names = 1, check.names = FALSE)
data2 <- read.csv(file2, row.names = 1, check.names = FALSE)

# Ensure the datasets contain the same samples
if (nrow(data1) != nrow(data2)) {
  stop("Datasets must have the same number of samples.")
}

if (!identical(rownames(data1), rownames(data2))) {
  stop("Sample identifiers or sample order do not match between datasets.")
}

# ------------------------------------------------------------
# Pearson correlation
# ------------------------------------------------------------

cor_matrix <- cor(
  data1,
  data2,
  method = "pearson",
  use = "pairwise.complete.obs"
)

# ------------------------------------------------------------
# Calculate correlation P-values
# ------------------------------------------------------------

get_p_value <- function(x, y) {
  
  complete <- complete.cases(x, y)
  
  if (sum(complete) < 3) {
    return(NA_real_)
  }
  
  if (sd(x[complete]) == 0 || sd(y[complete]) == 0) {
    return(NA_real_)
  }
  
  cor.test(
    x[complete],
    y[complete],
    method = "pearson"
  )$p.value
}

p_values_matrix <- matrix(
  NA_real_,
  nrow = ncol(data1),
  ncol = ncol(data2),
  dimnames = list(colnames(data1), colnames(data2))
)

for (i in seq_len(ncol(data1))) {
  for (j in seq_len(ncol(data2))) {
    p_values_matrix[i, j] <- get_p_value(
      data1[, i],
      data2[, j]
    )
  }
}

# ------------------------------------------------------------
# Convert to long format
# ------------------------------------------------------------

corr_long <- melt(
  cor_matrix,
  varnames = c("Feature_1", "Feature_2"),
  value.name = "correlation"
)

p_values_long <- melt(
  p_values_matrix,
  varnames = c("Feature_1", "Feature_2"),
  value.name = "p_value"
)

corr_long$p_value <- p_values_long$p_value

corr_long$correlation_label <- round(
  corr_long$correlation,
  2
)

# ------------------------------------------------------------
# Significance annotation
# ------------------------------------------------------------

corr_long$significance <- ifelse(
  corr_long$p_value < 0.001,
  "***",
  ifelse(
    corr_long$p_value < 0.01,
    "**",
    ifelse(
      corr_long$p_value < 0.05,
      "*",
      ""
    )
  )
)

# ------------------------------------------------------------
# Heatmap
# ------------------------------------------------------------

heatmap_plot <- ggplot(
  corr_long,
  aes(
    x = Feature_2,
    y = Feature_1,
    fill = correlation
  )
) +
  geom_tile(color = "white") +
  
  geom_text(
    aes(
      label = ifelse(
        significance != "",
        correlation_label,
        ""
      )
    ),
    size = 3,
    vjust = -0.3,
    fontface = "bold"
  ) +
  
  geom_text(
    aes(label = significance),
    size = 3,
    vjust = 1.3,
    fontface = "bold"
  ) +
  
  scale_fill_gradient2(
    low = "blue",
    mid = "white",
    high = "red",
    midpoint = 0,
    limits = c(-1, 1),
    name = "Pearson\nCorrelation"
  ) +
  
  labs(
    x = "Dataset 2 features",
    y = "Dataset 1 features"
  ) +
  
  scale_x_discrete(position = "top") +
  
  theme_minimal() +
  theme(
    axis.text.x = element_text(
      angle = 290,
      vjust = 1,
      hjust = 1,
      size = 10,
      face = "bold"
    ),
    axis.text.y = element_text(
      angle = 0,
      vjust = 1,
      hjust = 1,
      size = 10,
      face = "bold"
    ),
    axis.title = element_text(
      size = 12,
      face = "bold"
    )
  )

print(heatmap_plot)

# ------------------------------------------------------------
# Save correlation results
# ------------------------------------------------------------

write.csv(
  corr_long,
  "correlation_results_long.csv",
  row.names = FALSE
)

write.csv(
  cor_matrix,
  "correlation_matrix.csv",
  row.names = TRUE
)

write.csv(
  p_values_matrix,
  "correlation_pvalues.csv",
  row.names = TRUE
)

# ------------------------------------------------------------
# Significant correlations for Circos plot
# ------------------------------------------------------------

cor_df <- corr_long[
  !is.na(corr_long$p_value) &
    corr_long$p_value < 0.05,
]

# Remove correlations equal to zero
cor_df <- cor_df[
  cor_df$correlation != 0,
]

# ------------------------------------------------------------
# Circos plot
# ------------------------------------------------------------

if (nrow(cor_df) > 0) {
  
  cor_df$Feature_1 <- as.character(cor_df$Feature_1)
  cor_df$Feature_2 <- as.character(cor_df$Feature_2)
  
  # Link colour based on correlation direction
  cor_df$link_color <- ifelse(
    cor_df$correlation > 0,
    adjustcolor("red", alpha.f = 0.4),
    adjustcolor("blue", alpha.f = 0.4)
  )
  
  # Link width based on correlation strength
  cor_df$link_width <- abs(cor_df$correlation) * 10
  
  # Preserve the two feature sets as separate sector groups
  row_features <- unique(cor_df$Feature_1)
  col_features <- unique(cor_df$Feature_2)
  
  all_sectors <- c(
    row_features,
    col_features
  )
  
  # Larger gap between the two datasets
  gaps <- c(
    rep(2, max(length(row_features) - 1, 0)),
    15,
    rep(2, max(length(col_features) - 1, 0)),
    15
  )
  
  # Plot settings
  par(mar = c(0, 0, 0, 4))
  
  circos.clear()
  
  circos.par(
    start.degree = 90,
    gap.after = gaps,
    track.margin = c(0.01, 0.01),
    canvas.xlim = c(-1.2, 1.2),
    canvas.ylim = c(-1.2, 1.2)
  )
  
  chordDiagram(
    cor_df[, c(
      "Feature_1",
      "Feature_2",
      "correlation"
    )],
    col = cor_df$link_color,
    order = all_sectors,
    transparency = 0.4,
    link.lwd = cor_df$link_width,
    annotationTrack = "grid",
    preAllocateTracks = 1
  )
  
  # Add sector labels
  circos.trackPlotRegion(
    track.index = 1,
    panel.fun = function(x, y) {
      
      circos.text(
        CELL_META$xcenter,
        CELL_META$ylim[1],
        CELL_META$sector.index,
        facing = "clockwise",
        niceFacing = TRUE,
        adj = c(0, 0.5),
        cex = 0.8
      )
    },
    bg.border = NA
  )
  
  circos.clear()
  
} else {
  
  message(
    "No correlations met the P < 0.05 threshold; ",
    "Circos plot was not generated."
  )
}

# ------------------------------------------------------------
# Optional wide-format correlation matrix
# ------------------------------------------------------------

corr_wide <- dcast(
  corr_long,
  Feature_1 ~ Feature_2,
  value.var = "correlation"
)

write.csv(
  corr_wide,
  "correlation_matrix_wide.csv",
  row.names = FALSE
)
